import yfinance as yf
import pandas as pd
import numpy as np

def calculate_momentum_strategy(ticker_symbol):
    # 1. Fetch Intraday Data (5-minute intervals)
    ticker = yf.Ticker(ticker_symbol)
    df = ticker.history(period="5d", interval="5m")
    
    if df.empty:
        return None

    # 2. Basic Metrics: Gap and RVOL
    prev_close = ticker.history(period="2d")['Close'].iloc[-2]
    current_open = df['Open'].iloc[0]
    
    # Gap Percentage Calculation
    # $$\text{Gap \%} = \frac{\text{Open} - \text{Prev Close}}{\text{Prev Close}} \times 100$$
    gap_pct = ((current_open - prev_close) / prev_close) * 100
    
    # Relative Volume (Current Vol / 20-period Avg Vol)
    avg_vol = ticker.history(period="20d")['Volume'].mean()
    current_vol = df['Volume'].sum()
    rvol = current_vol / avg_vol

    # 3. Technical Indicators
    # VWAP Calculation: $$\text{VWAP} = \frac{\sum (\text{Price} \times \text{Volume})}{\sum \text{Volume}}$$
    df['TP'] = (df['High'] + df['Low'] + df['Close']) / 3
    df['VWAP'] = (df['TP'] * df['Volume']).cumsum() / df['Volume'].cumsum()
    
    # EMAs
    df['EMA9'] = df['Close'].ewm(span=9, adjust=False).mean()
    df['EMA20'] = df['Close'].ewm(span=20, adjust=False).mean()

    # 4. Bull Flag Detection Logic
    # Identify the 'Pole': A sharp move up on high volume
    recent_moves = df['Close'].pct_change(periods=3)
    has_pole = recent_moves.iloc[-5:-2].max() > 0.02 # 2% move in 15 mins
    
    # Identify the 'Flag': Consolidation (lower highs/lower lows) on decreasing volume
    is_consolidating = (df['High'].iloc[-3:] < df['High'].iloc[-4:-1]).all()
    vol_dropping = (df['Volume'].iloc[-3:] < df['Volume'].iloc[-4:-1]).all()

    # 5. Signal Logic
    is_above_vwap = df['Close'].iloc[-1] > df['VWAP'].iloc[-1]
    ema_crossover = df['EMA9'].iloc[-1] > df['EMA20'].iloc[-1]

    results = {
        "Ticker": ticker_symbol,
        "Gap %": round(gap_pct, 2),
        "RVOL": round(rvol, 2),
        "Above VWAP": is_above_vwap,
        "EMA Alignment": ema_crossover,
        "Bull Flag Setup": has_pole and is_consolidating and vol_dropping,
        "Action": "WATCH / BUY" if (gap_pct > 4 and is_above_vwap and has_pole) else "IGNORE"
    }
    
    return results

# Example Execution
tickers = ["AAPL", "TSLA", "NVDA", "AMD"] # Replace with your daily gap scanner results
for t in tickers:
    print(calculate_momentum_strategy(t))
